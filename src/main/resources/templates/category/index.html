''} `; tbody.appendChild(tr); }); } function openCreate(){ currentId = null; document.getElementById('modalTitle').textContent = 'New Category'; document.getElementById('name').value = ''; document.getElementById('description').value = ''; modal.show(); } async function edit(id){ const rs = await fetch(`${api}/${id}`); const x = await rs.json(); currentId = id; document.getElementById('modalTitle').textContent = 'Edit Category'; document.getElementById('name').value = x.name; document.getElementById('description').value = x.description ??
